% Greek language support.
%
% pandoc 3.7 changed how it wires up babel: where 3.1.3 emitted
% \babelprovide[main,import]{greek} (which reads babel's locale/el/babel-el.ini),
% it now passes `greek` as a documentclass option, which makes babel look for
% greek.ldf -- the babel-greek package, which the pandoctex image does not ship.
% The build dies with "Package babel Error: Unknown option 'greek'".
%
% The Makefile therefore passes `-V babel-lang=` to suppress the class option while
% keeping `lang=el` (so pdflang metadata survives), and we request the language here
% the way pandoc used to. This is what makes the TOC read "Περιεχόμενα" and figure
% labels "Σχήμα" rather than English. babel is already loaded by pandoc's template
% at the point this file is included.
\babelprovide[main,import]{greek}

% Long lines inside code blocks (terminal transcripts) would otherwise run off the
% right margin and get clipped. fvextra wraps them and marks the
% continuation with a hook arrow.

\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},breaklines=true,breakanywhere=true}
\DefineVerbatimEnvironment{verbatim}{Verbatim}{breaklines=true,breakanywhere=true}

% Branded masthead at the top of the first page, above the chapter's own H1.
% Suppressed for the combined notes.pdf, which uses a full cover page instead
% (tex/cover.tex defines \nomasthead). \ifdefined is used rather than a \newif flag
% because pandoc emits -V header-includes *before* this file, so a flag declared here
% could not be set from the command line.
\AtBeginDocument{%
  \ifdefined\nomasthead\else
  \thispagestyle{fancy}%
  \noindent
  \begin{minipage}[c]{0.68\linewidth}%

  \end{minipage}\hfill
  \begin{minipage}[c]{0.17\linewidth}%
    \raggedleft\includegraphics[width=\linewidth]{dit.jpg}%
  \end{minipage}%
  \par\vspace{6pt}%
  \noindent\textcolor{ditcyan}{\rule{\linewidth}{1.2pt}}%
  \par\vspace{2pt}%
  \noindent{\footnotesize\textcolor{ditcharcoal}{%
    Τμήμα Πληροφορικής και Τηλεπικοινωνιών \quad\textbullet\quad \coursename}}%
  \par\vspace{4pt}%
  \noindent\textcolor{ditcharcoal}{\rule{\linewidth}{0.4pt}}%
  \par\vspace{18pt}%
  \fi
}

% The notes mark significant spaces in printf output with ␣ (U+2423), a glyph Noto
% Mono does not have. Draw it instead, one monospace character wide.
\usepackage{newunicodechar}
\newunicodechar{␣}{{\sbox0{x}\makebox[\wd0]{\rule{0.4pt}{0.6ex}\rule{\dimexpr\wd0-2pt\relax}{0.4pt}\rule{0.4pt}{0.6ex}}}}

% pandoc writes a ^ inside inline code as \^{}, which under xelatex's TU encoding is
% the combining circumflex U+0302 - also missing from Noto Mono, so the caret in
% `[^...]` and `x ^ y` silently disappeared. An empty-argument \^ is a caret.
% Installed at \begin{document} because packages loaded after this file redefine \^.
\AtBeginDocument{%
  \let\notesorigcirc\^%
  \DeclareRobustCommand{\^}[1]{\ifx\relax#1\relax\textasciicircum\else\notesorigcirc{#1}\fi}%
}
